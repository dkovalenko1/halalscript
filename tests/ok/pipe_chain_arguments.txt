assalamu alaikum
# x |> f |> g(a) is g(f(x), a): the value of the stages before is the first
# argument, the arguments in brackets follow it.
dua dirham add(dirham x, dirham y) yalla
    ameen x + y
khalas
dua dinar scale(dinar x, dirham by) yalla
    ameen x * by
khalas
dua fatwa big(dinar x) yalla
    ameen x > 1000
khalas
# The pipe binds loosest of all: 2 + 3 is piped, not 3 alone.
bayan(2 + 3 |> add(10))
# A dirham stage feeds a dinar parameter: it widens, as in any call.
bismillah dinar wide := 1000000 |> add(1) |> scale(5000)
bayan(wide)
# A comparison needs brackets around the pipe.
bayan((wide |> big) == halal)
ameen 7 |> add(1) |> add(2)
