# c is read only by b, and b only by a: removing a leaves b unread, then c.
bismillah dirham c := 3
bismillah dirham b := c * 2
bismillah dirham a := b + 1
# A counter read only by its own assignments is read by no one.
inshallah dirham rounds := 0
kull dirham i := 1 ila 3 yalla
    rounds := rounds + i
khalas
# The inner total shadows the outer one and is never read: only it goes.
inshallah dirham total := 10
istikhara total > 5 yalla
    inshallah dirham total := 0
    total := total + 1
    bayan(1)
khalas
wallahi dirham UNUSED := 5
ameen total
