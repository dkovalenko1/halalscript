# c is read only by b, and b only by a: removing a leaves b unread, then c.
bismillah dirham c := 3
bismillah dirham b := c
bismillah fatwa a := b > 2
# The inner total shadows the outer one and is never read: it goes with its
# assignment, and the istikhara stays with the rest of its block.
inshallah dirham total := 10
istikhara total > 5 yalla
    inshallah dirham total := 0
    total := 1
    bayan(1)
khalas
